Target the CSS unrestricted / weight-8 frontier using the established pair-partition CPM construction of Okada and Kasai (arXiv:2607.14091), extending the short-cycle exponent search used for the author's earlier [[568,146,18]] entry. EX = [[13,82,43,52,73,10,49,97],

      [25,95,9,6,93,0,23,74]]

      [101,108,22,105,83,36,59,41],
      [101,97,58,39,60,33,72,76]]
def lift(E):
    H = np.zeros((3*P,8*P), dtype=np.int8)
    r = np.arange(P)
    for i in range(3):
        for j in range(8):
            H[i*P+r,j*P+(r-E[i][j])%P] = 1
    return H
HX, HZ = lift(EX), lift(EZ)
assert not np.any((HX @ HZ.T) % 2)
