Reproduction

The construction, in full (block-circulant parents over C16 with the paper's Table S5 shift lists):

import numpy as np

def blk(s, m):
    v = np.zeros(m, dtype=np.int8)
    for t in (s if isinstance(s, list) else [s]):
        v[t % m] ^= 1
    return np.array([np.roll(v, i) for i in range(m)])

def gala_abelian(L, J, m, F, G):
    h = L // 2
    FX = np.zeros((h*m, h*m), dtype=np.int8)
    GX = np.zeros((h*m, h*m), dtype=np.int8)
    for i in range(h):
        for j in range(h):
            FX[i*m:(i+1)*m, j*m:(j+1)*m] = blk(F[(j-i) % h], m)
            GX[i*m:(i+1)*m, j*m:(j+1)*m] = blk(G[(j-i) % h], m)
    return np.hstack([FX, GX])[:J*m], np.hstack([GX.T, FX.T])[:J*m]

HX, HZ = gala_abelian(L=12, J=5, m=16,
                      F=[10, 3, 2, 15, 12, 6],
                      G=[6, 10, 4, 1, 14, 13])
